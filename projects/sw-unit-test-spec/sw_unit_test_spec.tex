\section{Introduction}

\subsection{Purpose}
This document specifies the unit tests for the software unit
\texttt{\UnitName}. It defines the test approach, the test environment and
the individual test cases used to verify that the unit fulfils its
software detailed design and the allocated software requirements.

\subsection{Test Case Derivation}
Test cases are derived using:
\begin{itemize}[nosep]
  \item analysis of requirements and detailed design,
  \item equivalence class partitioning of inputs and states,
  \item boundary value analysis (min, min$-1$, max, max$+1$, zero, typical),
  \item error guessing based on experience and known defect classes,
  \item \ph{state transition / decision table analysis, if applicable}.
\end{itemize}

\subsection{Structural Coverage Targets}
\begin{tabularx}{\linewidth}{@{}X c c@{}}
  \toprule
  \textbf{Metric} & \textbf{Target} & \textbf{Justification if not met} \\ \midrule
  Function coverage  & 100\,\% & required \\
  Statement coverage & \ph{100\,\%} & \ph{review of uncovered lines} \\
  Branch coverage    & \ph{100\,\%} & \ph{review of uncovered branches} \\
  MC/DC              & \ph{n.a. / 100\,\%} & \ph{required for ASIL D} \\
  \bottomrule
\end{tabularx}

\subsection{Entry and Exit Criteria}
\textbf{Entry:}
\begin{itemize}[nosep]
  \item Unit source is baselined (\UnitVersion) and compiles without warnings.
  \item Detailed design [R2] and requirements [R1] are reviewed.
  \item Static analysis completed with no open blocking findings.
\end{itemize}
\textbf{Exit:}
\begin{itemize}[nosep]
  \item All specified test cases executed; all passed or deviations accepted.
  \item Coverage targets reached or gaps justified.
  \item Every requirement allocated to the unit is covered by $\geq 1$ test case.
\end{itemize}

\subsection{Pass/Fail Criteria}
A test case passes if every step yields the expected result
\ph{(within the stated tolerance)} and no unexpected side effect (assertion,
memory error, sanitizer report, timeout) occurs. Otherwise it fails.

\section{Test Cases}

\subsection{Test Case Template}
\ph{Remove this subsection once the first real test cases exist.}
Fields: \emph{Objective} -- what is verified; \emph{Requirements} -- traced
requirement IDs; \emph{Method} -- derivation and verification method;
\emph{Preconditions} -- state and configuration before the test;

\emph{Test steps} -- actions with expected results;
\emph{Postconditions} -- state after the test / cleanup.

\subsection{Nominal Behaviour}

\begin{testcase}{TC-UNIT-001}{Initialisation sets default state}
  \tcfield{Objective}{Verify that \texttt{Unit\_Init()} sets all internal
    state to the documented defaults.}
  \tcfield{Requirements}{SWR-0101, SWR-0102}
  \tcfield{Method}{Requirements-based; analysis of requirements}
  \tcfield{Priority}{High}
  \tcfield{Preconditions}{Unit not initialised; NvM mock returns
    \texttt{E\_OK}.}
  \tcfield{Input data}{none}
  \tcfield{Test steps}{%
    \begin{tcsteps}
      \tcstep{Call \texttt{Unit\_Init()}.}{Returns \texttt{E\_OK}.}
      \tcstep{Read \texttt{Unit\_GetState()}.}{Returns \texttt{UNIT\_IDLE}.}
      \tcstep{Check NvM mock call log.}{\texttt{NvM\_ReadBlock} called exactly once.}
    \end{tcsteps}}
  \tcfield{Postconditions}{Unit in \texttt{UNIT\_IDLE}.}
  \tcfield{Implementation}{\texttt{test/unit\_test.cpp: InitTest.SetsDefaults}}
\end{testcase}

\subsection{Boundary Values}

\begin{testcase}{TC-UNIT-002}{Setpoint accepted at upper limit}
  \tcfield{Objective}{Verify the setpoint range check at its upper bound.}
  \tcfield{Requirements}{SWR-0110}
  \tcfield{Method}{Boundary value analysis}
  \tcfield{Priority}{High}
  \tcfield{Preconditions}{TC-UNIT-001 preconditions; unit initialised.}
  \tcfield{Input data}{%
    \begin{tabular}[t]{@{}lll@{}}
      \texttt{setpoint} & 100 & max (valid)\\
      \texttt{setpoint} & 101 & max$+1$ (invalid)
    \end{tabular}}
  \tcfield{Test steps}{%
    \begin{tcsteps}
      \tcstep{\texttt{Unit\_SetSetpoint(100)}}{Returns \texttt{E\_OK}; setpoint = 100.}
      \tcstep{\texttt{Unit\_SetSetpoint(101)}}{Returns \texttt{E\_NOT\_OK}; setpoint unchanged (100).}
    \end{tcsteps}}
  \tcfield{Postconditions}{Setpoint = 100.}
  \tcfield{Implementation}{\ph{test file / test name}}
\end{testcase}
